```typescript
// src/lib/study-order.ts

type Schedulable = { next_review_at: string | null };

/**
 * SM-2 study order: overdue cards first (most overdue at the front),
 * then never-reviewed cards, then cards not yet due (soonest first).
 * Stable for equal keys; does not mutate the input.
 */
export function sortForStudy<T extends Schedulable>(cards: T[], now: Date = new Date()): T[] {
  const rank = (c: T): number => {
    if (c.next_review_at === null) return 1; // new
    return new Date(c.next_review_at) <= now ? 0 : 2; // due : not yet due
  };
  return [...cards].sort((a, b) => {
    const ra = rank(a);
    const rb = rank(b);
    if (ra !== rb) return ra - rb;
    if (a.next_review_at === null || b.next_review_at === null) return 0;
    return new Date(a.next_review_at).getTime() - new Date(b.next_review_at).getTime();
  });
}
```
